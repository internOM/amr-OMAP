<script>
/* ---------------- ROS ---------------- */

const rosStatus = document.getElementById("ros-status");

const ros = new ROSLIB.Ros({
    url: 'ws://192.168.251.47:9090'
});

ros.on('connection', () => {
    rosStatus.textContent = "CONNECTED";
    rosStatus.style.color = "#39FF14";
});

ros.on('close', () => {
    rosStatus.textContent = "DISCONNECTED";
    rosStatus.style.color = "#FF073A";
    publishCmdVel(0, 0);
});

const cmdVelTopic = new ROSLIB.Topic({
    ros: ros,
    name: '/cmd_vel',
    messageType: 'geometry_msgs/Twist'
});

function publishCmdVel(forward, turn) {
    const msg = new ROSLIB.Message({
        linear: { x: forward, y: 0, z: 0 },
        angular: { x: 0, y: 0, z: turn }
    });
    cmdVelTopic.publish(msg);
}

/* ---------------- JOYSTICK ---------------- */

const area = document.getElementById("joystick-area");
const knob = document.getElementById("knob");
const output = document.getElementById("direction");

let dragging = false;

function resetJoystick() {
    knob.style.left = "50%";
    knob.style.top = "50%";
    knob.style.transform = "translate(-50%, -50%)";
    output.textContent = "Center";
    publishCmdVel(0, 0);
}

knob.addEventListener("pointerdown", () => dragging = true);

window.addEventListener("pointerup", () => {
    dragging = false;
    resetJoystick();
});

area.addEventListener("pointermove", (e) => {
    if (!dragging) return;

    const rect = area.getBoundingClientRect();
    const radius = rect.width / 2;
    const knobRadius = knob.offsetWidth / 2;

    let x = e.clientX - rect.left - radius;
    let y = e.clientY - rect.top - radius;

    const dist = Math.sqrt(x*x + y*y);
    const maxDist = radius - knobRadius;

    if (dist > maxDist) {
        x = (x / dist) * maxDist;
        y = (y / dist) * maxDist;
    }

    knob.style.left = (radius + x) + "px";
    knob.style.top  = (radius + y) + "px";

    const forward = Math.max(-1, Math.min(1, -y / maxDist));
    const turn    = Math.max(-1, Math.min(1,  x / maxDist));

    output.textContent =
        `Forward: ${forward.toFixed(2)}  Turn: ${turn.toFixed(2)}`;

    publishCmdVel(forward, turn);
});

resetJoystick();
</script>
