<script>
    /* =========================
       ROS CONNECTION
       ========================= */
    const ros = new ROSLIB.Ros({
        url: 'ws://localhost:9090'   // change IP when serving from phone
    });

    const rosDot  = document.getElementById("ros-dot");
    const rosText = document.getElementById("ros-text");

    ros.on('connection', () => {
        rosDot.style.background = "green";
        rosText.textContent = "ROS: Connected";
    });

    ros.on('error', () => {
        rosDot.style.background = "red";
        rosText.textContent = "ROS: Error";
    });

    ros.on('close', () => {
        rosDot.style.background = "red";
        rosText.textContent = "ROS: Disconnected";
    });

    const autoModeTopic = new ROSLIB.Topic({
        ros: ros,
        name: '/auto_mode',
        messageType: 'std_msgs/Bool'
    });

    /* =========================
       AUTO MODE LOGIC
       ========================= */
    let isAuto = false;

    function toggleAuto() {
        isAuto = !isAuto;

        document.getElementById("output").innerText =
            isAuto ? "In auto mode" : "Auto mode off";

        const light = document.getElementById("auto-light");
        const btn   = document.getElementById("auto-btn");

        if (isAuto) {
            light.style.backgroundColor = "#39FF14";
            light.style.boxShadow = "0 0 15px #39FF14";
            btn.classList.add("active");
        } else {
            light.style.backgroundColor = "#FF073A";
            light.style.boxShadow = "0 0 15px #FF073A";
            btn.classList.remove("active");
        }

        autoModeTopic.publish(
            new ROSLIB.Message({ data: isAuto })
        );
    }

    /* =========================
       TIME + BATTERY (unchanged)
       ========================= */
    function updateTime() {
        document.getElementById("time").innerText =
            new Date().toLocaleTimeString();
    }
    setInterval(updateTime, 1000);
    updateTime();

    let batteryLevel = 100;
    function updateBattery() {
        batteryLevel -= 5;
        if (batteryLevel < 0) batteryLevel = 100;
        document.getElementById("battery-level").innerText = batteryLevel + "%";
    }
    setInterval(updateBattery, 2500);
    updateBattery();
</script>
